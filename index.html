Setup Guide

1

Create a Google Sheet

Go to sheets.google.com and create a new spreadsheet named "Meeting Scheduler".

2

Open Apps Script

In the sheet menu, click Extensions → Apps Script. This opens the script editor in a new tab.

3

Paste This Code

Delete everything in the script editor. Then copy and paste the code below:

// === Voice Meeting Scheduler - Google Apps Script === // Paste this in Extensions > Apps Script inside your Google Sheet const SHEET_NAME = 'Meetings'; function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['Timestamp','Title','Date','Time','Duration (min)','Your Email','Guest Email','Description','Status']); sheet.getRange(1, 1, 1, 9).setFontWeight('bold'); } // 1. Save to Google Sheet sheet.appendRow([ new Date().toISOString(), data.title, data.date, data.time, data.duration, data.yourEmail, data.guestEmail, data.description, 'Scheduled' ]); // 2. Create Google Calendar event with guest var startTime = new Date(data.date + 'T' + data.time + ':00'); var endTime = new Date(startTime.getTime() + parseInt(data.duration) * 60000); var event = CalendarApp.getDefaultCalendar().createEvent( data.title, startTime, endTime, { description: data.description + '\n\nScheduled via Voice Meeting Scheduler', guests: data.guestEmail, sendInvites: true } ); // 3. Send confirmation emails var details = 'Meeting: ' + data.title + '\nDate: ' + data.date + '\nTime: ' + data.time + '\nDuration: ' + data.duration + ' minutes' + '\nYour Email: ' + data.yourEmail + '\nGuest Email: ' + data.guestEmail + '\nDescription: ' + data.description + '\n\nCalendar event created.'; MailApp.sendEmail({ to: data.yourEmail, subject: 'Meeting Scheduled: ' + data.title, body: details }); MailApp.sendEmail({ to: data.guestEmail, subject: 'Meeting Invitation: ' + data.title, body: 'You have been invited to a meeting:\n\n' + details }); return ContentService.createTextOutput( JSON.stringify({ status: 'success', eventId: event.getId() }) ).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput( JSON.stringify({ status: 'error', message: error.toString() }) ).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); var headers = data[0]; var rows = data.slice(1).map(function(row) { var obj = {}; headers.forEach(function(h, i) { obj[h] = row[i]; }); return obj; }); return ContentService.createTextOutput(JSON.stringify(rows)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput( JSON.stringify({ status: 'error', message: error.toString() }) ).setMimeType(ContentService.MimeType.JSON); } }
4

Deploy as Web App

Click Deploy → New deployment.

  • • Click the gear icon next to "Select type" → choose Web app
  • • Description: Meeting Scheduler
  • • Execute as: Me
  • • Who has access: Anyone

Click Deploy. Google will ask you to authorize — click Advanced → Go to project to approve.

5

Copy the Web App URL

After deploying, you'll see a Web app URL. Copy it and paste it in the Settings panel below.

6

Paste URL in Settings

Close this guide, then paste the URL into the Apps Script URL field in Settings. You're all set!
